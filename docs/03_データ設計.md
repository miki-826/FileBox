# 03. データ設計

## 1. 保存先の全体像
| ストア | 場所 | 内容 | 書込権限 |
|---|---|---|---|
| メモリ索引 | Indexer プロセス内 | 名前索引・Trigram・容量集計（02 §3.2） | Indexer |
| スナップショット | `%PROGRAMDATA%\FileBox\index\<VolumeSerial>.snap` | メモリ索引の永続化（バイナリ、zstd圧縮） | Indexer |
| system.db | `%PROGRAMDATA%\FileBox\system.db` | アカウント・ロール・権限・クォータ・監査・ブックマーク | Indexer（管理者操作を代行） |
| user.db | `%LOCALAPPDATA%\FileBox\filebox.db` | ユーザーごとの学習・ワークスペース・設定・検索履歴 | App |

SQLite は WAL モード、`foreign_keys=ON`。マイグレーションは `schema_version` テーブルで管理。

## 2. system.db

```sql
-- アカウント
CREATE TABLE users (
  id            INTEGER PRIMARY KEY,
  windows_sid   TEXT NOT NULL UNIQUE,   -- 識別キー（Windows サインインをそのまま利用。パスワードは保持しない）
  account_name  TEXT NOT NULL,          -- 表示用 DOMAIN\user（起動時に最新化）
  display_name  TEXT,
  role_id       INTEGER NOT NULL REFERENCES roles(id),
  quota_bytes   INTEGER,                -- NULL=ロールの既定
  is_disabled   INTEGER NOT NULL DEFAULT 0,
  is_orphaned   INTEGER NOT NULL DEFAULT 0, -- Windows 側でアカウント削除済み
  created_at    TEXT NOT NULL,          -- 初回起動時に自動登録
  last_seen_at  TEXT
);

CREATE TABLE roles (
  id            INTEGER PRIMARY KEY,
  name          TEXT NOT NULL UNIQUE,   -- admin / standard / guest / custom...
  default_flags INTEGER NOT NULL,       -- 権限ビット（下記）
  quota_bytes   INTEGER                 -- NULL=無制限
);

-- フォルダ単位の権限 (subject はユーザーかロールのどちらか)
CREATE TABLE permissions (
  id          INTEGER PRIMARY KEY,
  user_id     INTEGER REFERENCES users(id) ON DELETE CASCADE,
  role_id     INTEGER REFERENCES roles(id) ON DELETE CASCADE,
  path        TEXT NOT NULL,            -- 正規化済みフルパス（小文字・末尾\なし）
  allow_flags INTEGER NOT NULL DEFAULT 0,
  deny_flags  INTEGER NOT NULL DEFAULT 0,
  CHECK ((user_id IS NULL) <> (role_id IS NULL))
);
CREATE INDEX ix_permissions_path ON permissions(path);
-- 権限ビット: 1=Read 2=Create 4=Modify 8=Delete 16=Share 32=Admin

-- クォータゾーン
CREATE TABLE quota_zones (
  id            INTEGER PRIMARY KEY,
  path          TEXT NOT NULL,          -- 対象フォルダ or ボリュームルート
  limit_bytes   INTEGER NOT NULL,
  scope         TEXT NOT NULL CHECK (scope IN ('all','user','role')),
  user_id       INTEGER REFERENCES users(id) ON DELETE CASCADE,
  role_id       INTEGER REFERENCES roles(id) ON DELETE CASCADE,
  warn_percent  INTEGER NOT NULL DEFAULT 80,
  os_enforced   INTEGER NOT NULL DEFAULT 0   -- NTFS クォータ連動
);

-- FileBox 経由で作成したファイルの作成者（ユーザー別容量計算用）
CREATE TABLE file_owners (
  volume_serial INTEGER NOT NULL,
  file_id       BLOB    NOT NULL,       -- 128bit FILE_ID_INFO
  user_id       INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  PRIMARY KEY (volume_serial, file_id)
);

-- ブックマーク（filebox://bookmark/<token>）
CREATE TABLE bookmarks (
  id            INTEGER PRIMARY KEY,
  token         TEXT NOT NULL UNIQUE,   -- 推測困難なランダム文字列
  volume_serial INTEGER NOT NULL,
  file_id       BLOB NOT NULL,
  last_path     TEXT NOT NULL,
  label         TEXT,
  created_by    INTEGER REFERENCES users(id),
  created_at    TEXT NOT NULL
);

-- 監査ログ
CREATE TABLE audit_log (
  id        INTEGER PRIMARY KEY,
  at        TEXT NOT NULL,
  user_id   INTEGER,
  action    TEXT NOT NULL,              -- save/move/copy/delete/rename/link_open/user_register/admin_verify/perm_change/quota_change
  src_path  TEXT,
  dst_path  TEXT,
  bytes     INTEGER,
  result    TEXT NOT NULL,              -- ok / denied / quota_exceeded / error
  detail    TEXT
);
CREATE INDEX ix_audit_at ON audit_log(at);

-- USN 追従状態
CREATE TABLE usn_state (
  volume_serial INTEGER PRIMARY KEY,
  journal_id    INTEGER NOT NULL,
  last_usn      INTEGER NOT NULL,
  snapshot_at   TEXT NOT NULL
);
```

## 3. user.db

```sql
-- 保存先学習
CREATE TABLE save_history (
  id          INTEGER PRIMARY KEY,
  tokens      TEXT NOT NULL,            -- 正規化トークン（スペース区切り、日付は <Y><M> に置換）
  ext         TEXT,
  source_app  TEXT,                     -- 保存元プロセス名
  folder_path TEXT NOT NULL,
  suggested_rank INTEGER,               -- 提案の何位を選んだか（NULL=提案外）
  chosen_at   TEXT NOT NULL
);
CREATE VIRTUAL TABLE save_history_fts USING fts5(tokens, content='save_history', content_rowid='id');

-- フォルダプロファイル（特徴語）
CREATE TABLE folder_profiles (
  folder_path TEXT NOT NULL,
  token       TEXT NOT NULL,
  tf          REAL NOT NULL,
  PRIMARY KEY (folder_path, token)
) WITHOUT ROWID;
CREATE INDEX ix_profiles_token ON folder_profiles(token);
CREATE TABLE token_df (token TEXT PRIMARY KEY, df INTEGER NOT NULL) WITHOUT ROWID;

-- 受け取りボックス
CREATE TABLE inbox_sources (id INTEGER PRIMARY KEY, path TEXT NOT NULL UNIQUE, enabled INTEGER NOT NULL DEFAULT 1);
CREATE TABLE inbox_rules (
  id          INTEGER PRIMARY KEY,
  name_pattern TEXT,                    -- ワイルドカード
  ext         TEXT,
  source_app  TEXT,
  target_path TEXT NOT NULL,
  auto_move   INTEGER NOT NULL DEFAULT 0,
  priority    INTEGER NOT NULL DEFAULT 100
);

-- ワークスペース（タブ・ペイン構成を JSON で保持）
CREATE TABLE workspaces (
  id         INTEGER PRIMARY KEY,
  name       TEXT NOT NULL,
  layout_json TEXT NOT NULL,
  updated_at TEXT NOT NULL
);

CREATE TABLE favorites   (id INTEGER PRIMARY KEY, path TEXT NOT NULL, label TEXT, sort INTEGER);
CREATE TABLE saved_searches (id INTEGER PRIMARY KEY, name TEXT NOT NULL, query TEXT NOT NULL);
CREATE TABLE search_history (id INTEGER PRIMARY KEY, query TEXT NOT NULL, at TEXT NOT NULL);
CREATE TABLE recent_items (path TEXT PRIMARY KEY, last_access TEXT NOT NULL, count INTEGER NOT NULL);
CREATE TABLE settings (key TEXT PRIMARY KEY, value TEXT NOT NULL);

-- エクスプローラー置き換え時に変更したレジストリの元の値（復元用）
CREATE TABLE shell_backup (reg_path TEXT NOT NULL, value_name TEXT NOT NULL, original_value TEXT, PRIMARY KEY (reg_path, value_name));
```

## 4. メモリ索引のエントリ構造（Indexer）
```csharp
[StructLayout(LayoutKind.Sequential, Pack = 4)]
struct IndexEntry            // 32 bytes
{
    public int  ParentId;    // EntryTable 内の親 index
    public int  NameOffset;  // NameHeap 内の位置
    public short NameLength;
    public EntryFlags Flags; // Directory, Hidden, System, Reparse
    public long Size;        // ディレクトリは配下合計（SizeAggregator）
    public long MTimeTicks;
    public int  FrnLow;      // MFT 参照番号の下位（ID 解決用ハッシュ表のキー）
}
```
- `Dictionary<ulong FRN, int id>`（USN イベントの親参照解決用）
- 削除は墓標フラグ → 定期的にコンパクション

## 5. データ保持・バックアップ
- audit_log は既定 1 年保持（設定可）、それ以前は月次でアーカイブ
- system.db は変更時に世代バックアップ（直近 7 世代）
- スナップショットは 10 分ごと＋サービス停止時に保存
